    function detectInAppBrowser(userAgent) {
        if (!userAgent || userAgent.match(/iPhone|iPad|iPod|Android/i) === null) {
            return {
                isInAppBrowser: false,
                iabHost: undefined
            };
        }
        var hosts = [
        {
            host: 'instagram',
            tokens: [/Instagram/i]
        },
        {
            host: 'tiktok',
            tokens: [/musical_ly/i, /\btrill_/i, /BytedanceWebview/i]
        },
        {
            host: 'facebook',
            tokens: [/FB_IAB/i, /FBAN/i, /FBAV/i, /FBIOS/i, /\bFB4A\b/i]
        },
        {
            host: 'snapchat',
            tokens: [/Snapchat/i]
        },
        {
            host: 'line',
            tokens: [/\bLine\//i]
        },
        {
            host: 'kakaotalk',
            tokens: [/KAKAOTALK/i]
        }
        ];
        for (var i = 0; i < hosts.length; i++) {
            var tokens = hosts[i].tokens;
            for (var j = 0; j < tokens.length; j++) {
                if (userAgent.match(tokens[j]) !== null) {
                    return {
                        isInAppBrowser: true,
                        iabHost: hosts[i].host
                    };
                }
            }
        }
        return {
            isInAppBrowser: false,
            iabHost: undefined
        };
    }

    (function() {
        if (window.cspBlobProbeInstalled) {
            return;
        }
        window.cspBlobProbeInstalled = true;

        var MAX_TRACKED_URLS = 100;
        var createStacks = {};
        var setStacks = {};
        var trackedCreateCount = 0;
        var trackedSetCount = 0;
        var reportedAssignment = false;
        var reportedViolation = false;
        var pageLoadId = Math.random().toString(36).slice(2, 10);

        // Same secret-path-segment pattern the Datadog beforeSend scrubber uses
        // (shared/lib/redact), embedded at build time because this inline script
        // runs before any bundle.
        var sensitivePathPattern = new RegExp("(\\/join\\/|%2Fjoin%2F|%252Fjoin%252F|\\/multiplayer\\/invite\\/|%2Fmultiplayer%2Finvite%2F|%252Fmultiplayer%252Finvite%252F|\\/accept-invite\\/|%2Faccept-invite%2F|%252Faccept-invite%252F|\\/data\\/user\\/unsubscribe\\/|%2Fdata%2Fuser%2Funsubscribe%2F|%252Fdata%252Fuser%252Funsubscribe%252F)([^/?#\\s\"'&]+)", 'gi');

        function redactPaths(text) {
            return String(text).replace(sensitivePathPattern, function(match, prefix) {
                return prefix + '[REDACTED]';
            });
        }

        // Strip query strings and fragments from URLs in stack traces so tokens in
        // deep links cannot leak into logs. Keeps scheme, origin, path, and the
        // :line:col suffix, and keeps extension ids on chrome-extension: URLs
        // because identifying the injecting extension is the point of the probe.
        function sanitizeStack(stack) {
            if (!stack) {
                return '';
            }

            var cleaned = String(stack)
            .split('\n')
            .slice(0, 12)
            .join('\n')
            .replace(/(https?|chrome-extension|moz-extension|safari-web-extension):\/\/[^\s)'"]+/g, function(u) {
                var cut = u.search(/[?#]/);

                return cut === -1 ? u : u.slice(0, cut);
            });

            return redactPaths(cleaned);
        }

        function sanitizeUrl(url) {
            return redactPaths(String(url).replace(/[?#][\s\S]*$/, '')).slice(0, 150);
        }

        function send(body) {
            if (window.location.search || window.location.hash) {
                if (!window.fetch || !window.Request) {
                    return;
                }

                try {
                    var canOmitReferrer =
                    new window.Request(window.location.href, {
                        referrerPolicy: 'no-referrer'
                    }).referrerPolicy === 'no-referrer';
                    if (!canOmitReferrer) {
                        return;
                    }

                    var request = window.fetch('https://http-intake.logs.us5.datadoghq.com/api/v2/logs?dd-api-key=pub31a5047a3a4692afb84a423db984dc57', {

                    }
                } catch (_e) {}

                return;
            }

            try {
                var xhr = new XMLHttpRequest();
                xhr.open('POST', 'https://http-intake.logs.us5.datadoghq.com/api/v2/logs?dd-api-key=pub31a5047a3a4692afb84a423db984dc57', true);
                xhr.setRequestHeader('Accept', 'application/json');
                xhr.setRequestHeader('Content-Type', 'application/json');
                xhr.send(body);
            } catch (_e) {}
        }

        function sendProbe(details) {
            details.pageLoadId = pageLoadId;

            try {
                send(JSON.stringify({
                    ddsource: 'browser',
                    ddtags: 'env:production,version:eb521d8b',
                    message: 'CSP blob script detected',
                    service: 'csp-blob-probe',
                    cspBlobProbe: details,

                        useragent: window.navigator.userAgent
                    },
                    view: {
                        url: sanitizeUrl(window.location.pathname)
                    }
                }));
            } catch (_e) {}
        }

        function reportAssignment(kind, url, setStack) {
            if (!setStacks[url] && trackedSetCount < MAX_TRACKED_URLS) {
                setStacks[url] = setStack;
                trackedSetCount++;
            }

            if (reportedAssignment) {
                return;
            }
            reportedAssignment = true;

            sendProbe({
                kind: kind,
                blobUrl: sanitizeUrl(url),
                createStack: sanitizeStack(createStacks[url]),
                setStack: sanitizeStack(setStack)
            });
        }

        function onViolation(event) {
            try {
                if (String(event.blockedURI || '').indexOf('blob') !== 0) {
                    return;
                }
                if (reportedViolation) {
                    return;
                }
                reportedViolation = true;

                var target = event.target;
                var isElement = !!(target && target.nodeType === 1);
                var src = isElement && typeof target.src === 'string' ? target.src : '';

                sendProbe({
                    kind: 'violation',
                    blobUrl: sanitizeUrl(src),
                    effectiveDirective: String(event.effectiveDirective || ''),
                    disposition: String(event.disposition || ''),
                    sourceFile: event.sourceFile ? sanitizeUrl(event.sourceFile) : '',
                    createStack: sanitizeStack(createStacks[src]),
                    setStack: sanitizeStack(setStacks[src]),
                    element: isElement ? sanitizeStack(target.outerHTML).slice(0, 500) : '',
                    targetNode: String((target && target.nodeName) || ''),
                    parentNode: String((isElement && target.parentNode && target.parentNode.nodeName) || '')
                });
            } catch (_e) {}
        }

        try {
            if (document.addEventListener) {
                document.addEventListener('securitypolicyviolation', onViolation, true);
            }
        } catch (_e) {}

        try {
            var originalCreateObjectURL = window.URL && window.URL.createObjectURL;
            if (originalCreateObjectURL) {
                window.URL.createObjectURL = function(obj) {
                    var url = originalCreateObjectURL.apply(window.URL, arguments);
                    try {
                        if (trackedCreateCount < MAX_TRACKED_URLS) {
                            createStacks[url] = new Error('createObjectURL').stack;
                            trackedCreateCount++;
                        }
                    } catch (_e) {}

                    return url;
                };
            }

            var proto = window.HTMLScriptElement && window.HTMLScriptElement.prototype;
            var srcDescriptor = proto && Object.getOwnPropertyDescriptor(proto, 'src');
            if (srcDescriptor && srcDescriptor.set) {
                Object.defineProperty(proto, 'src', {
                    configurable: true,
                    enumerable: srcDescriptor.enumerable,
                    get: srcDescriptor.get,
                    set: function(value) {

                                reportAssignment('src-set', value, new Error('script src assigned').stack);
                            }
                        } catch (_e) {}

                        return srcDescriptor.set.call(this, value);
                    }
                });
            }

            var originalSetAttribute = proto && proto.setAttribute;
            if (originalSetAttribute) {
                proto.setAttribute = function(name, value) {
                    try {
                        if (
                        String(name).toLowerCase() === 'src' &&
                        typeof value === 'string' &&
                        value.indexOf('blob:') === 0
                        ) {
                            reportAssignment('set-attribute', value, new Error('script setAttribute').stack);
                        }
                    } catch (_e) {}

                    return originalSetAttribute.apply(this, arguments);
                };
            }
        } catch (_e) {}
    })();
